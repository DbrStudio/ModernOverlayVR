cmake_minimum_required(VERSION 3.18)
project(edmo_openxr VERSION 0.1.0 LANGUAGES CXX)
find_package(OpenXR CONFIG REQUIRED)
find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)
add_library(edmo_openxr MODULE layer/layer.cpp layer/panel.cpp layer/frames.cpp)
target_compile_features(edmo_openxr PRIVATE cxx_std_17)
target_link_libraries(edmo_openxr PRIVATE OpenXR::headers Vulkan::Vulkan Threads::Threads nlohmann_json::nlohmann_json)
set_target_properties(edmo_openxr PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/layer")
target_compile_options(edmo_openxr PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
target_link_options(edmo_openxr PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/layer/exports.map" "LINKER:-z,defs")
set_property(TARGET edmo_openxr APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/layer/exports.map")
configure_file(layer/edmo.json "${CMAKE_BINARY_DIR}/layer/edmo.json" COPYONLY)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
