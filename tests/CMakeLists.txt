set(EDMO_TEST_LOADER "" CACHE FILEPATH "OpenXR loader or OpenComposite vrclient.so to exercise")
if(NOT EDMO_TEST_LOADER)
    set(EDMO_TEST_LOADER "$<TARGET_FILE:OpenXR::openxr_loader>")
endif()
add_library(test_runtime MODULE runtime.cpp)
target_link_libraries(test_runtime PRIVATE OpenXR::headers)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime.json" CONTENT
    "{\"file_format_version\":\"1.0.0\",\"runtime\":{\"library_path\":\"$<TARGET_FILE:test_runtime>\"}}\n")
add_executable(loader_test loader_test.cpp test_gpu.cpp)
target_link_libraries(loader_test PRIVATE OpenXR::headers Vulkan::Vulkan ${CMAKE_DL_LIBS})
target_compile_features(loader_test PRIVATE cxx_std_17)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(case trace panel unsupported live live_settings)
    set(mode "${case}")
    set(extra_env "")
    if(case STREQUAL "live_settings")
        set(mode live)
        set(extra_env "EDMO_TEST_SETTINGS=1")
    endif()
    set(driver)
    if(mode STREQUAL "live")
        set(driver "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/live_driver.py")
    endif()
    add_test(NAME loader_${case} COMMAND ${driver} "$<TARGET_FILE:loader_test>" "${EDMO_TEST_LOADER}" "$<TARGET_FILE:test_runtime>" ${mode})
    set_tests_properties(loader_${case} PROPERTIES
        ENVIRONMENT "XR_RUNTIME_JSON=${CMAKE_CURRENT_BINARY_DIR}/runtime.json;XR_API_LAYER_PATH=${CMAKE_BINARY_DIR}/layer;XR_ENABLE_API_LAYERS=XR_APILAYER_EDMO_overlay;EDMO_XR_MODE=${mode};EDMO_XR_LOG=-;XDG_STATE_HOME=${CMAKE_CURRENT_BINARY_DIR}/state-${case};${extra_env}"
        TIMEOUT 30 SKIP_RETURN_CODE 77)
endforeach()
